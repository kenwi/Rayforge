#version 100
precision highp float;

varying vec2 fragTexCoord;
varying vec4 fragColor;

uniform sampler2D texture0;
uniform vec3 palette[37];
uniform float softened;
// WebGL 1 / GLSL ES 100 has no textureSize(); the CPU passes the heat map dimensions.
uniform vec2 heatSize;

const float MaxIndex = 36.0;

// WebGL 1 fragment shaders may not index a uniform array with a non-const
// expression. Looping with the loop variable as the index is the portable fix.
vec3 samplePalette(int index)
{
    vec3 color = palette[0];
    for (int i = 0; i < 37; i++)
    {
        if (i == index)
        {
            color = palette[i];
        }
    }
    return color;
}

vec3 paletteColor(vec2 uv)
{
    vec2 snapped = (floor(clamp(uv, vec2(0.0), vec2(1.0)) * heatSize) + 0.5) / heatSize;
    float raw = texture2D(texture0, snapped).r * 255.0;
    int index = int(floor(raw + 0.5));
    if (index < 0) {
        index = 0;
    }
    if (index > int(MaxIndex)) {
        index = int(MaxIndex);
    }
    return samplePalette(index);
}

void main()
{
    vec2 uv = clamp(fragTexCoord, vec2(0.0), vec2(1.0));
    vec2 tex = uv * heatSize - 0.5;
    vec2 f = fract(tex);
    vec2 texel = 1.0 / heatSize;
    vec2 origin = (floor(tex) + 0.5) / heatSize;

    if (softened < 0.5)
    {
        gl_FragColor = vec4(paletteColor(uv), 1.0);
        return;
    }

    vec3 c00 = paletteColor(origin);
    vec3 c10 = paletteColor(origin + vec2(texel.x, 0.0));
    vec3 c01 = paletteColor(origin + vec2(0.0, texel.y));
    vec3 c11 = paletteColor(origin + texel);
    // Blend palette colors across the cell so the upscale is soft, while the
    // heat values underneath stay the discrete scatter.
    gl_FragColor = vec4(mix(mix(c00, c10, f.x), mix(c01, c11, f.x), f.y), 1.0);
}
